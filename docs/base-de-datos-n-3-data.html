Resumen numerico de antes y despues

resumen_antes <- data.frame(
  n = length(datos$antes),
  media = mean(datos$antes),
  ds = sd(datos$antes),
  mediana = median(datos$antes),
  RIC = IQR(datos$antes),
  min = min(datos$antes),
  max = max(datos$antes)
)

resumen_despues <- data.frame(
  n = length(datos$despues),
  media = mean(datos$despues),
  ds = sd(datos$despues),
  mediana = median(datos$despues),
  RIC = IQR(datos$despues),
  min = min(datos$despues),
  max = max(datos$despues)
)
