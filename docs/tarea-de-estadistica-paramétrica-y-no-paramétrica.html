8.6.1 Omitimos los datos faltantes

empleados <- na.omit(df6$Empleados)
agricultores <- na.omit(df6$Agricultores)
no_expuestos <- na.omit(df6$NoExpuestos)
n1 <- length(empleados)     # tamaño de muestra n_1
n2 <- length(agricultores)  # tamaño de muestra n_2
n3 <- length(no_expuestos)  # tamaño de muestra n_3
N <- n1 + n2 + n3           # tamaño de muestral total
k <- ncol(df)               # número de poblaciones

8.6.2 Promedio de cada grupo K

xbar_1 <- mean(empleados)
xbar_2 <- mean(agricultores)
xbar_3 <- mean(no_expuestos)

8.6.3 Hallamos promedio global

xbar_total <- (sum(empleados)+sum(agricultores)+sum(no_expuestos))/N

8.6.4 Hallamos SST

SST <- sum((empleados - xbar_total)^2) +
  sum((agricultores - xbar_total)^2) +
  sum((no_expuestos - xbar_total)^2)

8.6.5 Hallamos SSA

SSA <- n1*((xbar_1 - xbar_total)^2) + n2*((xbar_1-xbar_total)^2) + n3*((xbar_1-xbar_total)^2)

8.6.6 Hallemos SSE

SSE = SST-SSA

8.6.7 MSA Y MSE

MSA = SSA /(k-1)
MSE = SSE /(N-k)

8.6.8 Estadistico F

F = MSA/MSE

8.6.9 F tabulado

alpha = 0.05
v1 = k-1
v2 = N-k

F_critico=qf(1-alpha, df1=v1, df2=v2)

8.6.10 p valor

pvalor=1-pf(F, v1, v2)
